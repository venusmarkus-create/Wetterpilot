export default { // 1. HTTP-Anfragen (Fetch): Für den Schnee-Abruf in WetterPilot & manuellen Push-Test async fetch(request, env, ctx) { const url = new URL(request.url); // CORS Preflight Handhabung (wichtig für Browser-Abfragen) if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }, }); } // ========================================== // A. HND Schnee-Abfrage für Zugspitze // ========================================== if (url.pathname === "/api/schnee-zugspitze") { try { const hndResponse = await fetch("https://www.hnd.bayern.de/schnee/isar/zugspitze-2250m-16005705/tabelle", { headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36" } }); const html = await hndResponse.text(); // Liest den ersten Tabellenwert für Gesamtschneehöhe aus const match = html.match(/]*class="[^"]*wert[^"]*"[^>]*>\s*(\d+)\s*<\/td>/i); const schneeCm = match ? parseInt(match[1], 10) : 4; return new Response(JSON.stringify({ schneeCm }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*", "Cache-Control": "public, max-age=600" } }); } catch (err) { return new Response(JSON.stringify({ schneeCm: 4, error: err.message }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } // ========================================== // B. Manueller Test-Trigger für Push-Nachrichten // Aufruf über: https://dein-worker.workers.dev/api/check-warnings // ========================================== if (url.pathname === "/api/check-warnings") { const result = await pruefeUndSendeWarnungen(env); return new Response(JSON.stringify(result), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } return new Response("WetterPilot Worker läuft aktiv.", { status: 200 }); }, // 2. Automatischer Cron-Trigger (z.B. alle 10 oder 15 Minuten) async scheduled(event, env, ctx) { ctx.waitUntil(pruefeUndSendeWarnungen(env)); } }; // ========================================== // Funktion: DWD Unwetter prüfen & OneSignal Push senden // ========================================== async function pruefeUndSendeWarnungen(env) { try { // 1. DWD Warnungen abrufen (Beispiel BrightSky API für München/Laim) const lat = 48.14; const lon = 11.50; const dwdRes = await fetch(`https://api.brightsky.dev/current_weather?lat=${lat}&lon=${lon}`); // Falls du direkt die DWD-Warnungs-API oder Brightsky Alerts nutzt: const alertsRes = await fetch(`https://api.brightsky.dev/alerts?lat=${lat}&lon=${lon}`); const alertsData = await alertsRes.json(); if (!alertsData.alerts || alertsData.alerts.length === 0) { return { status: "Keine aktiven Warnungen vorhanden" }; } const aktuelleWarnung = alertsData.alerts[0]; // 2. KV-Storage Check: Wurde diese Warnung schon gesendet? if (env.WETTER_KV) { const letzteId = await env.WETTER_KV.get("letzte_warnung_id"); if (letzteId === aktuelleWarnung.event_code) { return { status: "Warnung wurde bereits zuvor gepusht" }; } // Speichern damit nicht doppelt gesendet wird (für 4 Stunden) await env.WETTER_KV.put("letzte_warnung_id", aktuelleWarnung.event_code, { expirationTtl: 14400 }); } // 3. Push über OneSignal REST API verschicken const payload = { app_id: env.ONESIGNAL_APP_ID, included_segments: ["All"], // oder ["Total Subscriptions"] headings: { de: `⚠️ ${aktuelleWarnung.headline \vert{}\vert{} "Wetterwarnung"}` }, contents: { de: aktuelleWarnung.description \vert{}\vert{} "Neue amtliche Unwetterwarnung für deinen Standort." }, url: "https://venusmarkus-create.github.io/Wetterpilot/" }; const pushRes = await fetch("https://onesignal.com/api/v1/notifications", { method: "POST", headers: { "Content-Type": "application/json; charset=utf-8", "Authorization": `Basic ${env.ONESIGNAL_REST_API_KEY}` }, body: JSON.stringify(payload) }); const pushResult = await pushRes.json(); return { status: "Push verschickt", result: pushResult }; } catch (error) { return { status: "Fehler beim Ausführen", error: error.message }; } }
